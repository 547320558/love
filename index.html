import turtle import random # 定义祝福语句列表 blessings = [ "多喝热水", "好好爱自己", "梦想成真", "早点休息", "要快乐", "万事顺遂", "要平安", "保持好心情", "要好运", "天天开心", "笑口常开", "平安喜乐", "别熬夜", "要健康", "万事胜意" ] # 设置画布 screen = turtle.Screen() screen.setup(800, 600) screen.bgcolor("black") # 定义绘制卡片的函数 def draw_card(text, x, y, color): t = turtle.Turtle() t.hideturtle() t.speed(0) t.penup() t.goto(x, y) t.pendown() t.color(color) t.begin_fill() for _ in range(4): t.forward(100) t.left(90) t.end_fill() t.penup() t.goto(x + 20, y + 30) t.color("black") t.write(text, font=("Arial", 12, "normal")) # 随机绘制多个卡片 colors = ["#FFC0CB", "#ADD8E6", "#90EE90", "#F0E68C", "#FFA500", "#DDA0DD"] for _ in range(50): text = random.choice(blessings) x = random.randint(-350, 300) y = random.randint(-250, 200) color = random.choice(colors) draw_card(text, x, y, color) turtle.done()
